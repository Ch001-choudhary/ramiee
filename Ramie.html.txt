    <div class="container">
        <h1>Will you be my Valentine, Ramsha? 💝</h1>
        <p>We've had so many laughs together—let's make this day epic! 😘</p>
        <div class="buttons">
            <button class="yes" onclick="yesClick()" style="position: relative;">Yes! 🥰</button>
            <button class="no" id="noButton" onclick="noClick()">No way! 🙄</button>
        </div>
        <div class="message" id="message"></div>
    </div>

    <script>
        const noButton = document.getElementById('noButton');
        const message = document.getElementById('message');
        let dodgeCount = 0;
        
        // SUPER FAST RANDOM DODGING
        noButton.addEventListener('mouseover', function(e) {
            dodgeCount++;
            message.innerHTML = `Caught me ${dodgeCount} times! 😂 Keep trying!`;
            
            // Ultra-fast random teleportation
            const maxX = window.innerWidth - 120;
            const maxY = window.innerHeight - 80;
            
            // More aggressive random movement
            const dodgeX = Math.random() * maxX;
            const dodgeY = Math.random() * maxY;
            
            noButton.style.left = dodgeX + 'px';
            noButton.style.top = dodgeY + 'px';
            noButton.style.transform = 'translate(-50%, -50%)';
            
            // Extra shake effect
            noButton.style.animation = 'shake 0.05s infinite';
            
            // Reset shake after 0.5s
            setTimeout(() => {
                noButton.style.animation = 'shake 0.1s infinite';
            }, 500);
        });
        
        // Initial random position
        window.addEventListener('load', function() {
            const maxX = window.innerWidth - 120;
            const maxY = window.innerHeight - 80;
            noButton.style.left = (Math.random() * maxX) + 'px';
            noButton.style.top = (Math.random() * maxY) + 'px';
            noButton.style.transform = 'translate(-50%, -50%)';
        });
        
        // Handle window resize
        window.addEventListener('resize', function() {
            const maxX = window.innerWidth - 120;
            const maxY = window.innerHeight - 80;
            const rect = noButton.getBoundingClientRect();
            if (rect.right > window.innerWidth || rect.bottom > window.innerHeight) {
                noButton.style.left = (Math.random() * maxX) + 'px';
                noButton.style.top = (Math.random() * maxY) + 'px';
            }
        });
        
        function yesClick() {
            message.innerHTML = 'Yay I knew it! 🎉💕';
            noButton.style.display = 'none';
            createHearts();
        }
        
        function noClick() {
            message.innerHTML = 'Hehe, too slow Ramsha! 😜';
            // Immediately dodge again
            setTimeout(() => {
                const maxX = window.innerWidth - 120;
                const maxY = window.innerHeight - 80;
                noButton.style.left = (Math.random() * maxX) + 'px';
                noButton.style.top = (Math.random() * maxY) + 'px';
            }, 10);
        }
        
        function createHearts() {
            for (let i = 0; i < 50; i++) {
                setTimeout(() => {
                    const heart = document.createElement('div');
                    heart.innerHTML = '💖';
                    heart.style.position = 'fixed';
                    heart.style.left = Math.random() * 100 + '%';
                    heart.style.animation = `float ${Math.random() * 3 + 2}s linear infinite`;
                    heart.style.fontSize = '2em';
                    document.querySelector('.hearts').appendChild(heart);
                    setTimeout(() => heart.remove(), 5000);
                }, i * 100);
            }
        }
    </script>
